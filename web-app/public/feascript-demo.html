Drop-In Reusable ES Modules (examples/feascript-loop/js/)

import { OpenCadBridge } from './js/opencad-bridge.js';
import { cdmlToFEAScriptGeometry } from './js/cdml-to-feascript.js';
import { solveThermal2D, buildNearestWallFieldGrid } from './js/feascript-solver.js';

// 1. Convert any OpenCAD CDML <sketch> directly into FEAScript geometry & boundaries
const cad = await OpenCadBridge.create();
const pass1 = cad.evalCdml(cdmlXml);
const cdmlGeom = cdmlToFEAScriptGeometry(cdmlXml);

// 2. Solve 2D PDE in FEAScript & inject nearest-wall 3D voxel field into OpenCAD
const fem = solveThermal2D(cdmlGeom, params);
cad.writeFieldBuffer(buildNearestWallFieldGrid(fem, pass1.bbox, params));
const pass2 = cad.evalCdml(cdmlXml); // Evaluates B-Rep + <probe id="bore_temp">
