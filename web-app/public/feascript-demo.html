// 2. Minimal pure-ESM OpenCAD + FEAScript 0.4.1 closed loop (with SharedArrayBuffer):
import { FEAScriptModel, evaluateSolution } from "https://core.feascript.com/dist/feascript.esm.js";

const mem = new WebAssembly.Memory({ initial: 128, maximum: 512, shared: true });
const wasmBytes = await fetch("https://cdn.jsdelivr.net/gh/hassaanmaqsood/opencad@dist/cad_core.wasm").then(r => r.arrayBuffer());
const { instance } = await WebAssembly.instantiate(wasmBytes, {
  env: { memory: mem },
  opencad: { field_alloc: () => 0, field_eval_scalar: (_s,_x,_y,_z,d) => d, field_eval_vec3: () => {}, cmd_write: () => {} }
});
instance.exports.kernel_init();
// 1) Evaluate CDML -> 3D B-Rep + 2D profile with named edges ("bottom", "right", "top", "left", "bore")
// 2) Solve 2D thermal field in FEAScriptModel("heatConduction")
// 3) Sample evaluateSolution() onto a 32x24x4 Float32Array at memory offset 0x1F0000
// 4) Re-evaluate CDML so field_scalar(0, x, y, z, 25.0) drives OpenCAD probes & geometry!
